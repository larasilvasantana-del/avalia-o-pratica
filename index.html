const scoreDisplay = document.getElementById('score'); let lassoPosition = 0; let direction = 1; let gameInterval = null; let isRunning = false; let score = 0; // Alternar entre modo Claro e Escuro themeToggleBtn.addEventListener('click', () => { const currentTheme = document.body.getAttribute('data-theme');
