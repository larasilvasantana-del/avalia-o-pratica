Lógica JavaScript (script.js) JavaScript // --- 1. Alternador de Tema (Claro / Escuro) --- const themeBtn = document.getElementById('theme-btn'); themeBtn.addEventListener('click', () => { const currentTheme = document.body.getAttribute('data-theme'); if (currentTheme === 'dark') { document.body.removeAttribute('data-theme'); themeBtn.textContent = 'Modo Escuro'; } else { document.body.setAttribute('data-theme', 'dark'); themeBtn.textContent = 'Modo Claro'; } }); // --- 2. Simulador de Tempo de Laçada --- const lacoBtn = document.getElementById('laco-btn'); const resultado = document.getElementById('resultado'); let emAndamento = false; let tempoInicio; let intervalo; lacoBtn.addEventListener('click', () => { if (!emAndamento) { // Inicia o cronômetro emAndamento = true; tempoInicio = Date.now(); lacoBtn.textContent = 'Laçar!'; lacoBtn.style.backgroundColor = '#2e7d32'; // Verde intervalo = setInterval(() => { const decorrido = ((Date.now() - tempoInicio) / 1000).toFixed(2); resultado.textContent = `Tempo: ${decorrido}s`; }, 10); } else { // Para o cronômetro emAndamento = false; clearInterval(intervalo); const tempoFinal = ((Date.now() - tempoInicio) / 1000).toFixed(2); lacoBtn.textContent = 'Iniciar Nova Laçada'; lacoBtn.style.backgroundColor = ''; // Volta ao CSS padrão if (tempoFinal < 4.0) { resultado.textContent = `Tempo: ${tempoFinal}s — Excelente laçada! Tempo de campeão!`; } else { resultado.textContent = `Tempo: ${tempoFinal}s — Boa prova! Tente ser mais rápido na próxima.`; } } });
